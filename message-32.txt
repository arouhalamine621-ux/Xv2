    local function matchesSelectedPet(prompt)
        if not PState then
            return false

        if not sel then
            return false
        end
        local model = prompt:FindFirstAncestorOfClass("Model")
        if not model then
            return false
        end

            if not plotNode then

    local function isPromptValid(prompt, targetPos)
        if not prompt or not prompt.Parent then
            return false
        end
        if not prompt.Enabled then
            return false
        end
        local promptPos = getPromptPos(prompt)
        if not promptPos then
            return false
        end
        local model = prompt:FindFirstAncestorOfClass("Model")
        if model then
            local plotsFolderRef = workspace:FindFirstChild("Plots")
            if plotsFolderRef then
                local ancestorModel = prompt:FindFirstAncestorWhichIsA("Model")
                while ancestorModel and ancestorModel.Parent ~= plotsFolderRef do
                    ancestorModel = ancestorModel.Parent
                end
                if ancestorModel then
                    local plotSign = ancestorModel:FindFirstChild("PlotSign")
                    if plotSign then
                        local surfGui = plotSign:FindFirstChildWhichIsA("SurfaceGui", true)
                        local textLabel = surfGui and surfGui:FindFirstChildWhichIsA("TextLabel", true)
                        if textLabel then

                                or txt:find(game.Players.LocalPlayer.DisplayName:lower(), 1, true)
                            then
                                return false
                            end
                        end
                    end
                end
            end
        end
        if _G.NEAREST_INSTANT_MODE == true then
            local rIdx = getPlotIndex(targetPos)
            local pIdx = getPlotIndex(promptPos)
            if not rIdx or rIdx ~= pIdx then
                return false
            end
        end
        if _G.NEAREST_INSTANT_MODE ~= true then
            if not matchesSelectedPet(prompt) then
                return false
            end
        end
        local maxDist = typeof(prompt.MaxActivationDistance) == "number"
                and prompt.MaxActivationDistance > 0
                and prompt.MaxActivationDistance
            or C_Auto.RADIUS
        local effectiveRadius = math.min(C_Auto.RADIUS, maxDist)
        return (promptPos - targetPos).Magnitude <= effectiveRadius
    end
    local function checkCooldown(prompt, cooldown)
        local now = os.clock()
        local last = promptCooldowns[prompt]
        if last and now - last < cooldown then
            return false
        end
        promptCooldowns[prompt] = now
        return true
    end
    local function firePrompt(prompt, count, cooldown)
        if not prompt or not prompt.Parent then
            return
        end
        if not prompt.Enabled then
            return
        end
        if not checkCooldown(prompt, cooldown) then
            return
        end
        for _ = 1, count do
            pcall(function()
                fireproximityprompt(prompt, 0)
            end)
        end
    end
    local function setupPrompt(prompt)
        if promptMap[prompt] then
            return
        end
        promptMap[prompt] = true
        local function checkAndFire()
            local root = getRootPart()
            if not root then
                return
            end
            local pos = root.Position
            if isPromptValid(prompt, pos) then
                C_Auto.AUTO_STEAL = true
                local now = os.clock()
                local lastTime = promptTimes[prompt]
                if not lastTime or now - lastTime >= 0.04 then
                    promptTimes[prompt] = now
                    firePrompt(prompt, 25, 0)
                end
            end
        end
        task.defer(function()
            checkAndFire()
        end)
        pcall(function()
            prompt:GetPropertyChangedSignal("Enabled"):Connect(function()
                if prompt.Enabled then
                    checkAndFire()
                end
            end)
        end)
        prompt.AncestryChanged:Connect(function()
            if not prompt:IsDescendantOf(workspace) then
                promptMap[prompt] = nil
                promptCooldowns[prompt] = nil
                promptTimes[prompt] = nil
            end
        end)
    end
    local function scanPlots()
        local plotsFolderRef = workspace:FindFirstChild("Plots")
        if not plotsFolderRef then
            return
        end
        for _, plot in ipairs(plotsFolderRef:GetChildren()) do
            local podiums = plot:FindFirstChild("AnimalPodiums")
            if podiums then
                for _, desc in ipairs(podiums:GetDescendants()) do
                    if desc:IsA("ProximityPrompt") then
                        setupPrompt(desc)
                    end
                end
            end
        end
    end
    scanPlots()
    workspace.DescendantAdded:Connect(function(desc)
        if desc:IsA("ProximityPrompt") and desc:FindFirstAncestor("AnimalPodiums") then
            setupPrompt(desc)
        end
    end)
    task.spawn(function()
        while task.wait(_G.getSafePollRate()) do
            local root = getRootPart()
            if not root then
                C_Auto.AUTO_STEAL = false
                continue
            end
            local pos = root.Position
            local activeFound = false
            for prompt in pairs(promptMap) do
                if isPromptValid(prompt, pos) then
                    activeFound = true
                    if C_Auto.AUTO_STEAL then
                        firePrompt(prompt, 5, 0.05)
                    end
                end
            end
            C_Auto.AUTO_STEAL = activeFound
        end
    end)
end

function DeepCopy(tbl)
    if type(tbl) ~= "table" then
        return tbl
    end
    local copy = {}
    for k, v in pairs(tbl) do
        copy[k] = DeepCopy(v)
    end
    return copy
end

function MergeDefaults(target, defaults)
    for k, v in pairs(defaults) do
        if type(v) == "table" then
            if type(target[k]) ~= "table" then
                target[k] = DeepCopy(v)
            else
                MergeDefaults(target[k], v)
            end
        elseif target[k] == nil then
            target[k] = v
        end
    end
end

            y.MenuKey = i.SavedKeybinds.MenuKey
            writefile("SlicedzHub.json", HttpSvcRef:JSONEncode(y))
        end)
    end
end
if _needMigrationSave then
    saveConfigToFile()
end
do
    _G.StickyCarpetSpeed = i.CarpetSpeed == true
    _G.StickyCarpetSpeedValue = math.clamp(tonumber(i.CarpetSpeedValue) or 140, 20, 400)
    _G.StickyCarpetSpeedKeyName = type(i.CarpetSpeedKey) == "string" and i.CarpetSpeedKey or "Q"
    _G.StickyInfJump = i.InfJump ~= false
    _G.StickyAutoKickOnSteal = i.AutoKickOnSteal == true
    _G.StickyFOV = tonumber(i.FOV) or 80
    _G.StickyResetCooldown = tonumber(i.ResetCooldown) or 2.5
    _G.StickyResetFlingTime = tonumber(i.ResetFlingTime) or 5
    _G.StickyAntiBee = i.AntiBeeDisco ~= false
    _G.StickyAntiDieDisabled = i.AntiDie == false

    local function syncGlobalsToConfig()
        i.CarpetSpeedValue = math.clamp(tonumber(_G.StickyCarpetSpeedValue) or 140, 20, 400)
        i.FOV = tonumber(_G.StickyFOV) or i.FOV
        i.ResetCooldown = tonumber(_G.StickyResetCooldown) or i.ResetCooldown
        i.ResetFlingTime = tonumber(_G.StickyResetFlingTime) or i.ResetFlingTime
        i.AntiBeeDisco = _G.StickyAntiBee ~= false
        i.AntiDie = _G.StickyAntiDieDisabled ~= true
        if type(_G.StickyCarpetSpeedKeyName) == "string" then
            i.CarpetSpeedKey = _G.StickyCarpetSpeedKeyName
        end
        if _G.StickyInfJump ~= nil then i.InfJump = _G.StickyInfJump == true end
        if _G.StickyCarpetSpeed ~= nil then i.CarpetSpeed = _G.StickyCarpetSpeed == true end
        if _G.StickyAutoKickOnSteal ~= nil then i.AutoKickOnSteal = _G.StickyAutoKickOnSteal == true end
    end

    local lastSnapshot = nil
    _G.StickySaveConfigNow = function()
        pcall(function()
            syncGlobalsToConfig()
            local snap = HttpSvcRef:JSONEncode(i)
            if snap ~= lastSnapshot then
                lastSnapshot = snap
                saveConfigToFile()
            end
        end)
    end

    task.spawn(function()
        task.wait(5)
        while true do
            _G.StickySaveConfigNow()
            task.wait(5)
        end
    end)
end
do
    local ready = false
    local waiters = {}

    _G.StickyOnBoot = function(fn)
        if type(fn) ~= "function" then return end
        if ready then task.spawn(fn) return end
        waiters[#waiters + 1] = fn
    end

    task.spawn(function()
        local deadline = os.clock() + 30
        while not workspace:FindFirstChild("Plots") and os.clock() < deadline do
            task.wait(0.25)
        end
        local charDeadline = os.clock() + 15
        while os.clock() < charDeadline do
            local ch = LocalPlayerRef.Character
            if ch and ch:FindFirstChild("HumanoidRootPart") and ch:FindFirstChildOfClass("Humanoid") then
                break
            end
            task.wait(0.25)
        end
        task.wait(1.5)
        ready = true
        for _, fn in ipairs(waiters) do task.spawn(fn) end
        waiters = {}
    end)
end
do
    local Config = i
    local saveConfig = saveConfigToFile

    local function setToggle(name, enabled)
        local sync = _G.SlicedzSyncToggleUI
        if type(sync) == "function" then
            pcall(sync, name, enabled)
        end
    end

    if _G.StickyXray      == nil then _G.StickyXray      = true end
    if _G.StickyXrayAlpha == nil then _G.StickyXrayAlpha = 0.9  end

    ;(function()
        local FOLDERS = { "Base", "PlotSign", "FriendPanel", "Cash", "Laser",
            "Decorations", "Skin", "Unlock", "Purchases" }
        local orig = setmetatable({}, { __mode = "k" })
        local conns, gen = {}, 0

        local function paint(o, a)
            if not o:IsA("BasePart") then return end
            if orig[o] == nil then orig[o] = (o.Transparency == a) and 0 or o.Transparency end
            local base = orig[o]
            if base >= 1 then return end
            local want = base + (1 - base) * a
            if math.abs(o.Transparency - want) > 0.01 then o.Transparency = want end
        end

        local function calm()
            while _G.StickyStealHold do task.wait(0.15) end
        end

        local function track(root, a, id)
            if not root or id ~= gen then return end
            paint(root, a)
            local n = 0
            for _, d in ipairs(root:GetDescendants()) do
                if id ~= gen then return end
                paint(d, a)
                n = n + 1
                if n % 250 == 0 then task.wait() end
            end
            conns[#conns + 1] = root.DescendantAdded:Connect(function(d)
                if id == gen then paint(d, a) end
            end)
        end

        local function doPlot(plot, a, id)
            if not plot or id ~= gen then return end
            for _, fname in ipairs(FOLDERS) do
                if id ~= gen then return end
                track(plot:FindFirstChild(fname), a, id)
            end
            if id ~= gen then return end
            conns[#conns + 1] = plot.ChildAdded:Connect(function(c)
                if id ~= gen then return end
                for _, fname in ipairs(FOLDERS) do
                    if c.Name == fname then track(c, a, id) break end
                end
            end)
            local pods = plot:FindFirstChild("AnimalPodiums")
            if not pods then return end
            local function pod(pd)
                for _, c in ipairs(pd:GetChildren()) do
                    if c.Name == "Claim" then track(c, a, id)
                    elseif c.Name == "Base" then track(c:FindFirstChild("Decorations"), a, id) end
                end
            end
            for _, pd in ipairs(pods:GetChildren()) do pod(pd) end
            conns[#conns + 1] = pods.ChildAdded:Connect(function(pd)
                if id ~= gen then return end
                task.wait(0.1)
                if id == gen then pod(pd) end
            end)
        end

        local function stop()
            for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
            conns, gen = {}, gen + 1
        end

        function _G.StickyEnableXray()
            Config.XRay = true
            saveConfig()
            setToggle("XRay", true)
            setToggle("X-Ray", true)
            stop()
            local id = gen
            local a = math.clamp(tonumber(_G.StickyXrayAlpha) or 0.9, 0, 1)
            task.spawn(function()
                while id == gen and not workspace:FindFirstChild("Plots") do 
                    task.wait(0.5) 
                end
                local plots = workspace:FindFirstChild("Plots")
                if id ~= gen or not plots then return end
                calm()
                for _, p in ipairs(plots:GetChildren()) do
                    if id ~= gen then return end
                    pcall(doPlot, p, a, id)
                end
            end)
        end
    end)()
end
